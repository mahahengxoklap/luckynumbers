function playGame() { if (selectedNum === null) { alert("ກະລຸນາເລືອກຕົວເລກກ່ອນ!"); return; } const betAmount = parseInt(document.getElementById('betAmount').value); if (isNaN(betAmount) || betAmount <= 0) { alert("ກະລຸນາໃສ່ຈຳນວນເງິນວາງພັນໃຫ້ຖືກຕ້ອງ!"); return; } if (betAmount > currentCredit) { alert("ເຄຣດິດຂອງທ່ານບໍ່ພຽງພໍ!"); return; } const payload = { phone: currentPhone, chosenNumber: selectedNum, betAmount: betAmount }; fetch(SCRIPT_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify(payload) }) .then(res => res.json()) .then(data => { if (data.success) { currentCredit = data.newCredit; document.getElementById('userCredit').innerText = currentCredit.toLocaleString() + " ₭"; showPopup(data.isWin, selectedNum, data.winningNumber, betAmount, data.netWinAmount || (betAmount * 5)); } else { // ດຶງ Error ຕົວຈິງຈາກ Apps Script ມາໂຊໃຫ້ເຫັນເລີຍ alert("ຜິດພາດ: " + (data.error || "ບໍ່ทราบສາເຫດ")); } }) .catch(err => { alert("ເຊື່ອມຕໍ່ Server ບໍ່ໄດ້: " + err); }); }
